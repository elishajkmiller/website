(function(){

  if(!root){return;}
  function good(v){

    if(v.indexOf('HERE')>-1){return false;}
    return true;
  }
  var cardEls=root.querySelectorAll('.dja-card');
  var w;

    if(pic){
      pic.setAttribute('role','img');
      if(ttl){pic.setAttribute('aria-label',ttl.textContent);}
    }
    if(who){
      var hs=who.getAttribute('data-headshot');
      var site=who.getAttribute('data-site');
      var by=document.createElement('div');
      by.className='dja-by';
      if(good(hs)){
        var av=document.createElement('span');
        av.className='dja-av';
        av.style.setProperty('--a',"url('"+hs+"')");
        by.appendChild(av);
      }
      var nm=document.createElement('span');
      nm.textContent='by '+who.textContent;
      by.appendChild(nm);
      card.replaceChild(by,who);
      if(good(site)){
        var lnk=document.createElement('a');
        lnk.className='dja-site';
        lnk.href=site;
        lnk.textContent='Artist website \u2197';
        lnk.setAttribute('target','_blank');
        lnk.setAttribute('rel','noopener');
        card.appendChild(lnk);
      }
    }
    var bodyEl=document.createElement('div');
    bodyEl.className='dja-body';
    var kids=Array.prototype.slice.call(card.children);
    var q;
    for(q=0;q<kids.length;q++){
      if(kids[q]!==pic){bodyEl.appendChild(kids[q]);}
    }
    card.appendChild(bodyEl);

  var i;
  var step=Math.ceil(photos.length/8);
  for(i=0;i<photos.length;i+=step){
    var s=document.createElement('div');
    s.className='dja-s';
    s.style.backgroundImage=window.getComputedStyle(photos[i]).backgroundImage;
    bg.appendChild(s);
  }
  var track=root.querySelector('.dja-track');
  var k,r;
  for(r=0;r<2;r++){
    for(k=0;k<photos.length;k++){
      var d=document.createElement('div');
      d.className='dja-t';
      d.style.backgroundImage=window.getComputedStyle(photos[k]).backgroundImage;
      d.style.backgroundPosition=window.getComputedStyle(photos[k]).backgroundPosition;
      track.appendChild(d);
    }
  }
  var lb=document.createElement('div');
  lb.className='dja-lb';
  lb.hidden=true;
  lb.innerHTML='<b>&times;</b><img alt=""><p></p>';
  root.appendChild(lb);
  var lbi=lb.querySelector('img');
  var lbp=lb.querySelector('p');
  function openLb(n){
    var m=window.getComputedStyle(photos[n]).backgroundImage.match(/url\(["']?([^"')]+)["']?\)/);
    var card=photos[n].parentNode;
    lbi.src=m?m[1]:'';
    lbp.textContent=card.querySelector('h3').textContent+', '+card.querySelector('.dja-by span:last-child').textContent;
    lb.hidden=false;
  }
  root.addEventListener('click',function(e){
    var t=e.target.closest('.dja-t');
    var ph=e.target.closest('.dja-photo');
    if(t){openLb(Array.prototype.indexOf.call(track.children,t)%photos.length);return;}
    if(ph){openLb(Array.prototype.indexOf.call(photos,ph));return;}
    if(!lb.hidden){lb.hidden=true;}
  });
  document.addEventListener('keydown',function(e){if(e.key==='Escape'){lb.hidden=true;}});
  var slides=bg.children;
  var n=0;
  slides[0].className='dja-s dja-on';
  var calm=false;
  try{calm=window.matchMedia('(prefers-reduced-motion: reduce)').matches;}catch(e){}
  if(calm){return;}
  if(slides.length<2){return;}
  setInterval(function(){
    slides[n].className='dja-s';
    n=(n+1)%slides.length;
    slides[n].className='dja-s dja-on';
  },10000);
})();
