<script>
(function(){
  var root=document.getElementById('dja-art');
  if(!root){return;}
  var pics=root.querySelectorAll('.dja-photo');
  var y;
  for(y=0;y<pics.length;y++){
    var ttl=pics[y].parentNode.querySelector('h3');
    pics[y].setAttribute('role','img');
    if(ttl){pics[y].setAttribute('aria-label',ttl.textContent);}
  }
  var links=root.querySelectorAll('.dja-site');
  var heads=root.querySelectorAll('.dja-av');
  var z;
  for(z=0;z<links.length;z++){
    var h=links[z].getAttribute('href')||'';
    if(h===''||h.indexOf('HERE')>-1){
      links[z].parentNode.removeChild(links[z]);
    }else{
      links[z].textContent='Artist website \u2197';
      links[z].setAttribute('target','_blank');
      links[z].setAttribute('rel','noopener');
    }
  }
  for(z=0;z<heads.length;z++){
    var st=heads[z].getAttribute('style')||'';
    if(st.indexOf("url('')")>-1||st.indexOf('HERE')>-1){heads[z].parentNode.removeChild(heads[z]);}
  }
  var bg=root.querySelector('.dja-bg');
  var photos=root.querySelectorAll('.dja-photo');
  if(!bg){return;}
  if(photos.length===0){return;}
  var i;
  var step=Math.ceil(photos.length/8);
  for(i=0;i<photos.length;i+=step){
    var s=document.createElement('div');
    s.className='dja-s';
    s.style.backgroundImage=window.getComputedStyle(photos[i]).backgroundImage;
    bg.appendChild(s);
  }
  var track=root.querySelector('.dja-track');
  var k,r;
  for(r=0;r<2;r++){
    for(k=0;k<photos.length;k++){
      var d=document.createElement('div');
      d.className='dja-t';
      d.style.backgroundImage=window.getComputedStyle(photos[k]).backgroundImage;
      d.style.backgroundPosition=window.getComputedStyle(photos[k]).backgroundPosition;
      track.appendChild(d);
    }
  }
  var lb=root.querySelector('.dja-lb');
  var lbi=lb.querySelector('img');
  var lbp=lb.querySelector('p');
  function openLb(n){
    var m=window.getComputedStyle(photos[n]).backgroundImage.match(/url\(["']?([^"')]+)["']?\)/);
    var card=photos[n].parentNode;
    lbi.src=m?m[1]:'';
    lbp.textContent=card.querySelector('h3').textContent+', '+card.querySelector('.dja-by span:last-child').textContent;
    lb.hidden=false;
  }
  root.addEventListener('click',function(e){
    var t=e.target.closest('.dja-t');
    var ph=e.target.closest('.dja-photo');
    if(t){openLb(Array.prototype.indexOf.call(track.children,t)%photos.length);return;}
    if(ph){openLb(Array.prototype.indexOf.call(photos,ph));return;}
    if(!lb.hidden){lb.hidden=true;}
  });
  document.addEventListener('keydown',function(e){if(e.key==='Escape'){lb.hidden=true;}});
  var slides=bg.children;
  var n=0;
  slides[0].className='dja-s dja-on';
  var calm=false;
  try{calm=window.matchMedia('(prefers-reduced-motion: reduce)').matches;}catch(e){}
  if(calm){return;}
  if(slides.length<2){return;}
  setInterval(function(){
    slides[n].className='dja-s';
    n=(n+1)%slides.length;
    slides[n].className='dja-s dja-on';
  },10000);
})();
</script>
